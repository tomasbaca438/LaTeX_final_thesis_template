\newtermentry{viscosity}{
	name={Viscosity},
	description={A physical quantity expressing a fluid's resistance to deformation under shear stress. It manifests as internal friction.}
}

\newtermentry{diffusion}{
	name={Diffusion},
	description={A process in which particles spontaneously spread out in space due to their thermal motion.}
}

\newtermentry{capillarity}{
	name={Capillarity},
	description={A phenomenon in which a liquid rises or falls in a narrow capillary tube due to surface tension.}
}
